            deadlineMs: (this.options.terminationTermGraceMs ?? GW) + CUA_DEV_TIMEOUT_MS + (this.options.healthTimeoutMs ?? 5e3) + Zse
          }),
          c = s ?? {
            socketPath: n.socketPath,
            handle: a,
            oldTransportRetired: !1
          };
        this.pendingRefreshMarker = c, this.assertNotExternallyStopped(t), await this.stopCurrentHandle("permission refresh", {
          allowSigkill: !1,
          termGraceMs: this.options.terminationTermGraceMs ?? GW
        }), c.oldTransportRetired = !0, this.assertNotExternallyStopped(t);
        let d = await this.doStart(t, {
          helperAppPath: n.helperAppPath,
          socketPath: n.socketPath
        }, "restart");
        return await a.complete(), this.pendingRefreshMarker === c && (this.pendingRefreshMarker = null), this.assertNotExternallyStopped(t), d
      })().finally(() => {
        this.startInFlight === o && (this.startInFlight = null)
      });
      return this.startInFlight = o, o.then(s => ({
        handle: s,
        reused: !0
      }))
    }
    async finishPendingTermination(t, n, r = !1) {
      let o = this.pendingTermination;
      if (o) {
        await this.terminateHelperPid(o.pid, t, {
          socketPath: o.socketPath,
          helperAppPath: o.helperAppPath
        }, n ?? o.policy, r);
        return
      }
      await this.terminationInFlight
    }
    async recordAndCleanupFailedLaunch(t, n, r = null) {
      let o = this.pendingFailedLaunch;
      if (o && (o.socketPath !== t.socketPath || o.helperAppPath !== t.helperAppPath)) throw new CuaHelperError("termination_failed", "A previous failed ZCode Computer Use launch is still unresolved; refusing to replace its lifecycle blocker", {
        cause: n
      });
      this.pendingFailedLaunch ??= {
        ...t,
        context: `startup failed: ${n instanceof Error?n.message:String(n)}`,
        cancellationPublished: !1
      }, this.ensureFailedLaunchRevoked(this.pendingFailedLaunch);
      try {
        r != null && await this.terminateHelperPid(r, "live process identity verification failed", t), await this.finishPendingFailedLaunch("startup failure")
      } catch (s) {
        throw new CuaHelperError("termination_failed", `ZCode Computer Use startup failed and the launched process could not be conclusively cleaned up: ${s instanceof Error?s.message:String(s)}`, {
          cause: n
        })
      }
    }
    async finishPendingFailedLaunch(t) {
      if (!this.pendingFailedLaunch) {
        await this.failedLaunchCleanupInFlight;
        return
      }
      if (this.failedLaunchCleanupInFlight) {
        await this.failedLaunchCleanupInFlight, this.pendingFailedLaunch && await this.finishPendingFailedLaunch(t);
        return
      }
      let n = this.pendingFailedLaunch;
      this.ensureFailedLaunchRevoked(n);
      let o = this.runFailedLaunchCleanup(n, t).then(() => {
        this.pendingFailedLaunch === n && (this.pendingFailedLaunch = null)
      }).finally(() => {
        this.failedLaunchCleanupInFlight === o && (this.failedLaunchCleanupInFlight = null)
      });
      this.failedLaunchCleanupInFlight = o, await o
    }
    async runFailedLaunchCleanup(t, n) {
      let r = this.options.discoverLaunchedHelperProcesses ?? NW,
        o = Math.max(1, this.options.failedLaunchCleanupPollIntervalMs ?? 100),
        s = Math.max(1, Math.ceil(Math.max(0, this.options.failedLaunchCleanupQuietMs ?? 500) / o)),
        a = this.options.failedLaunchCleanupDelay ?? VW,
        c = this.options.failedLaunchCleanupNow ?? Date.now,
        d = 0,
        l = t.cancellationPublished ? 0 : Math.ceil(Math.max(0, t.guard.deadlineEpochMs - c()) / o),
        p = Math.max(32, s + 8, l + s + 8);
      for (let u = 0; u < p; u += 1) {
        let f = await r({
          socketPath: t.socketPath,
          helperAppPath: t.helperAppPath
        });
        if (f.state === "unknown") throw new CuaHelperError("termination_failed", `Cannot confirm failed ZCode Computer Use launch cleanup (${n}): ${f.detail??"process identity is unknown"}`);
        if (f.state === "observed") {
          d = 0;
          let g = [...new Set(f.pids)].filter(v => Number.isInteger(v) && v > 1);
          if (g.length === 0) throw new CuaHelperError("termination_failed", `Failed ZCode Computer Use launch discovery returned observed without a valid pid (${n})`);
          for (let v of g) await this.terminateHelperPid(v, t.context, {
            socketPath: t.socketPath,
            helperAppPath: t.helperAppPath
          })
        } else {
          d += 1;
          let g = t.cancellationPublished || c() >= t.guard.deadlineEpochMs;
          if (d >= s && g) return
        }
        await a(o)
      }
      throw new CuaHelperError("termination_failed", `Failed ZCode Computer Use launch did not reach a safely revoked quiet state (${n})${t.cancellationPublishError?`; cancel sentinel: ${t.cancellationPublishError}`:""}`)
    }
    ensureFailedLaunchRevoked(t) {
      if (!t.cancellationPublished) try {
        (this.options.publishBrokerLaunchCancellation ?? Tz)(t.socketPath, t.guard), t.cancellationPublished = !0, t.cancellationPublishError = void 0;
        try {
          (this.options.scheduleBrokerLaunchGuardCleanup ?? Mz)(t.guard)
        } catch (n) {
          this.options.logger?.warn(void 0, `failed to schedule cua helper launch-cancel cleanup: ${n instanceof Error?n.message:String(n)}`)
        }
      } catch (n) {
        t.cancellationPublishError = n instanceof Error ? n.message : String(n)
      }
    }
    async terminateHelperPid(t, n, r = {}, o = {}, s = !1) {
      if (t == null) return;
      if (this.pendingTermination && this.pendingTermination.pid !== t) throw new CuaHelperError("termination_failed", `Cannot terminate ZCode Computer Use pid ${t} (${n}): pid ${this.pendingTermination.pid} is still pending termination`);
      let a = !s && this.pendingTermination ? this.pendingTermination.policy : {
        allowSigkill: o(o as any).allowSigkill ?? !0,
        termGraceMs: o(o as any).termGraceMs ?? this.options.terminationTermGraceMs ?? 1e3
      };
      if (this.pendingTermination ??= {
          pid: t,
          context: n,
          ...r,
          policy: a
        }, this.terminationInFlight) {
        await this.terminationInFlight, this.pendingTermination?.pid === t && await this.terminateHelperPid(t, n, r, a, s);
        return
      }
      let d = this.runTerminationSequence(t, n, r, a).catch(l => {
        throw l instanceof CuaHelperError ? l : new CuaHelperError("termination_failed", `Failed to confirm ZCode Computer Use pid ${t} termination (${n}): ${l instanceof Error?l.message:String(l)}`, {
          cause: l
        })
      }).then(() => {
        this.pendingTermination?.pid === t && (this.pendingTermination = null)
      }).finally(() => {
        this.terminationInFlight === d && (this.terminationInFlight = null)
      });
      this.terminationInFlight = d, await d
    }
    async runTerminationSequence(t, n, r, o) {
      if (this.signalHelperIfStillOwned(t, "SIGTERM", n, r) && !await this.waitForHelperDeparture(t, o.termGraceMs, n, r)) {
        if (!o.allowSigkill) throw new CuaHelperError("termination_failed", `ZCode Computer Use pid ${t} remained alive after SIGTERM (${n}); SIGKILL is disabled for permission refresh`);
        if (this.signalHelperIfStillOwned(t, "SIGKILL", n, r) && !await this.waitForHelperDeparture(t, this.options.terminationKillGraceMs ?? 1e3, n, r)) throw new CuaHelperError("termination_failed", `ZCode Computer Use pid ${t} remained alive after SIGTERM and SIGKILL (${n})`)
      }
    }
    signalHelperIfStillOwned(t, n, r, o) {
      let s = this.readHelperPidEvidence(t, o);
      if (s.state === "unknown") throw new CuaHelperError("termination_failed", `Cannot safely signal ZCode Computer Use pid ${t} (${r}): ${s.reason??"process identity is unknown"}`);
      if (s.state !== "helper") {
        if (s.state === "unrelated") {
          let a = s.command ? ` (now: ${s.command})` : "";
          this.options.logger?.warn(void 0, `skipping ${n} of cua helper pid ${t} (${r}): ${s.reason??"pid no longer looks like our Helper"}${a}`)
        }
        return !1
      }
      try {
        (this.options.killProcess ?? jse)(t, n)
      } catch (a) {
        this.options.logger?.warn(void 0, `failed to send ${n} to cua helper pid ${t} (${r}): ${a instanceof Error?a.message:String(a)}`)
      }
      return !0
    }
    async waitForHelperDeparture(t, n, r, o) {
      let s = Math.max(1, this.options.terminationPollIntervalMs ?? 100),
        a = Math.max(1, Math.ceil(Math.max(0, n) / s)),
        c = this.options.terminationDelay ?? VW;
      for (let d = 0; d < a; d += 1) {
        await c(s);
        let l = this.readHelperPidEvidence(t, o);
        if (l.state === "unknown") throw new CuaHelperError("termination_failed", `Cannot confirm ZCode Computer Use pid ${t} departure (${r}): ${l.reason??"process identity is unknown"}`);
        if (l.state !== "helper") return !0
      }
      return !1
    }
    readHelperPidEvidence(t, n) {
      try {
        return this.collectHelperPidEvidence(t, n)
      } catch (r) {
        return {
          state: "unknown",
          reason: r instanceof Error ? r.message : String(r)
        }
      }
    }
    async restart() {
      return this.startInFlight ? this.startInFlight : this.beginFreshRestart(this.stopGeneration)
    }
    async restartAfterCurrentStart() {
      if (this.restartAfterCurrentStartInFlight) return this.restartAfterCurrentStartInFlight;
      let t = this.stopGeneration,
        r = (async () => {
          for (;;) {
            this.assertNotExternallyStopped(t);
            let s = this.startInFlight;
            if (!s) break;
            try {
              await s
            } catch {}
          }
          let o = await this.beginFreshRestart(t);
          return this.assertNotExternallyStopped(t), o
        })().finally(() => {
          this.restartAfterCurrentStartInFlight === r && (this.restartAfterCurrentStartInFlight = null)
        });
      return this.restartAfterCurrentStartInFlight = r, r
    }
    async restartAfterCurrentStartPreservingTransport(t = {}) {
      if (this.restartPreservingTransportInFlight) return this.restartPreservingTransportInFlight;
      let n = this.stopGeneration,
        o = (async () => {
          for (;;) {
            this.assertNotExternallyStopped(n);
            let c = this.startInFlight;
            if (!c) break;
            try {
              await c
            } catch {}
          }
          this.assertNotExternallyStopped(n);
          let s = await this.retryCompletedRefreshMarkerForLiveHandle();
          if (s) return s;
          let a = this.handle;
          if (!a) {
            t(t as any).beforeFreshStart?.();
            let c = await this.beginFreshRestart(n);
            return this.assertNotExternallyStopped(n), {
              handle: c,
              reused: !1
            }
          }
          return this.beginTransportPreservingRestart(n, a)
        })().finally(() => {
          this.restartPreservingTransportInFlight === o && (this.restartPreservingTransportInFlight = null)
        });
      return this.restartPreservingTransportInFlight = o, o
    }
