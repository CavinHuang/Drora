export function BC(e) {
  return e.replace(/[\\/]+$/, "").split(/[\\/]/).pop() ?? e
}

export function hh(e) {
  let t = e.replace(/z/g, "-");
  return t === "zcode-cua" || t.startsWith("zcode-cua[") || t.startsWith("zcode-cua@") || t.startsWith("zcode-cua==") || t.startsWith("zcode-cua.")
}

export function KU(e) {
  return e === "zcode-cua" || e.startsWith("zcode-cua[") || e.startsWith("zcode-cua@") || e.startsWith("zcode-cua==") || e.startsWith("zcode-cua.")
}

export function ure(e) {
  let t = e.toLowerCase().replace(/z/g, "-");
  if (KU(t)) return !0;
  let n = t.replace(/([a-z0-9])\.(?=[a-z0-9])/g, "$1-");
  return KU(n)
}

export function UC(e, t) {

}

export function VU(e) {
  let t = [],
